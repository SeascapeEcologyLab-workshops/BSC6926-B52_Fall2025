dlogisticD = function(K, rd, N_0, years) {
  logis = tibble(Nt = NA, time = 0:years)
  logis$Nt[logis$time == 0] = N_0
  
  for(i in 1:(length(logis$time)-1)){
    logis$Nt[i+1] = logis$Nt[i]  + rd*logis$Nt[i]*(1 - logis$Nt[i] / K)
  }
  return(logis)
}

dirand = function(l_mean, l_sd, N_0, years) {
  pop = tibble(Nt = NA, time = 0:years) |> 
    mutate(Nt = if_else(time == 0, N_0, Nt),
           lambda = truncnorm::rtruncnorm(n(), mean = l_mean, sd = l_sd, a=0, b=Inf))
  
  for(i in 2:nrow(pop)){
    pop$Nt[i] = pop$lambda[i-1] * pop$Nt[i-1]
  }

  return(pop)
}
